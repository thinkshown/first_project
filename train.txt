- console
mkdir - создать каталог
cd - перейти в каталог (cd .git/)
cd - - перейти в предыдущую локацию
ls - вывести содержимое каталога
ls -lah - вывести содержимое каталога (l - в табличном виде, a - включая скрытые папки, h - в человекочитаемом размере)
pwd - путь к текущей локации
cat - вывести содержимое файла (cat config)
rm <file_name> - удалить файл
touch - создает новый пустой файл
mv <old_file_name> <new_file_name> - переименовать файл
cp <file_name> <path> - перенос файла

- git
git - вызов краткой справки
history - история команд
git init создать git-репозиторий в текущей папке

git config --global user.name "thinkshown" - установка пользователя для всех (глобал)
git config --show-origin --list - показывает в каких файлах хранятся все настройки
git config --global color.ui true - включение цветного вывода информации

git status - показывает изменение в рабочем каталоге и статус

git log - показывает историю коммитов 
git log --oneline - показывает историю коммитов в компактном виде
git log --stat - посмотреть статистику по коммитам
git log -p - просмотр патчей (деталей каждого изменения) по коммитам
git log -X - покажет Х последних коммитов
git log --graph - отображение веток в истории коммитов
git log --author="author_name" - фильтр коммитов по автору
git log --pretty=format:"spec_formats" - форматирование вывода лога
git log --since="2012-01-01" --until="2012-12-31" - фильтрация коммитов по датам "до" и "после"

git shortlog - компактный отчёт о коммитах в репозитории группирует коммиты по авторам и выводит список их имён вместе с количеством, сделанных ими коммитов и краткими описаниями этих коммитов
git shortlog -e - показывает почту рядом с именем автора коммита
git shortlog -n - сортирует вывод по количеству коммитов, от большего к меньшему
git shortlog -s - выводит только статистику (кол-во коммитов) без списка сообщений
git shortlog --since="2012-01-01" --until="2012-12-31" - фильтрация коммитов по датам "до" и "после"

git add <file_name> - добавить в индекс (staging area)
git add . - добавить все файлы
git add -f <file_name> - форсированно добавить файл (игнорируемый)

git commit - добавить коммит через редактор
git commit -a - добавить коммит со всеми изменениями
git commit --amend - изменить последний коммит
git commit -m - добавить коммит с сообщением
git commit -am - избавляемся от отдельной команды git add и задаем сообщение

git reset - сбросить текущий индекс
git reset <file_name> - исключить указанный файл из индекса
git reset --hard - отменяет все изменения, сделанные после последнего коммита
git reset HEAD~1 - переместить "голову" репозитория на 1 коммит назад (~1 - перемещение на 1 коммит назад от текущего положения головы) или HEAD^ (HEAD~3 == HEAD^^^). При это все изменения из коммита переходят в индекс
git reset ORIG_HEAD --hard - вернуться к коммиту, на который указывает ORIG_HEAD
git reset --soft - изменения из коммита перебрасываются в индекс
git reset --mixed (по умолчанию) - сбросит коммит и индекс (возврат к коммиту, на который указывает HEAD с очищением индекса и сохранением рабочей директории)

git revert HEAD - откатить последний коммит (создавая новый "обратный коммит")
git revert HEAD --no-edit - откатить последний коммит без ввода сообщения для коммита
git revert <commit_id> - откатить коммит по его идентификатору (хэш-коду)
git revert HEAD~3 - откатить третий коммит с конца
git revert --abort - отменить незавершенный откат коммита
git revert <commit_id1>..<commit_id2> - откатить серию коммитов
git revert <commit_id> -n, --no-commit - откатить коммит без созданий нового коммита (изменения попадают в индекс)

git reflog - показывает локальный журнал изменений указателя HEAD (сокращение от reference log - журнал ссылок)
git reflog expire --expire=7.days --all --verbose - удаление записей старше 7 дней (verbose - выводит подробный отчет о работе в терминал)

git gc --force - сборщик мусора (garbage collector)

git show - показывает изменения с предыдущего коммита
git show <commit_id> - вывести только изменения, указанные в этом коммите
git show <commit_id> --name-only - только имена файлов, измененных в коммите
git show <commit_id1>..<commit_id2> - вывести все коммиты с изменениями между двумя коммитами
git show HEAD~1:<file_path> - просмотр изменений файла в определенном коммите

git rm <file_name> - удаление файла из Git-репозитория
git rm <file_name> -f, --force - форсированное удаление (если есть изменения в индексе)
git rm <my_folder> -r, --recursive - удаление каталога

git clean - удалить untracked файлы
git clean -f - удалить принудительно (предотвращение случайного удаления файлов)
git clean -d - удалить папки (директории)
git clean -fd -n - "холостой прогон команды" (посмотреть, что будет)

git diff - вывод изменений после последнего коммита (не добавленных в индекс)
git diff --staged, --cached - вывод изменений, добавленных в индекс
git diff --staged --word-diff - форматирование изменений по словам (а не по строкам)
git diff --staged --w - исключение пустых строк из вывода
git diff HEAD - вывод изменений после последнего коммита (включая добавленные в индекс)
git diff <commit_id> - вывести все накопленные изменения между текущим состоянием и состоянием на момент указанного коммита
git diff <commit_id1> <commit_id2> - вывод разницы между двумя указанными коммитами (суммарно)
git diff HEAD~1 <file> - вывод изменений в конкретном файле после указанного коммита
git diff <file_1> <file_2> - сравнение двух произвольных файлов (даже вне git-репозитория)
git diff <commit_id> --name-only - вывести только список файлов, измененных с указанного коммита и до текущего состояния репозитория

git difftool <commit_id> - указание хэша коммита для вывода изменений после него

git mv <old_file_name> <new_file_name> - переименовать файл (перенести)

для игнора каких-то файлов нужно создать файл .gitignore и добавить в него названия файлов, которые нужно игнорить. если нужно добавить все файлы определенного разрешения, то *.log, если папку, то folder/*. Исключение - !<file_name>. можно взять типовой файл .gitignore для своего языка программирование на https://github.com/github/gitignore/tree/main или https://www.toptal.com/developers/gitignore и сделать его основным в конфиге с помощью следующих команд:
## глобальные настройки
# указываем глобальную настройку для пути к файлу .gitignore
git config --global core.excludesfile ~/.gitignore_global
# создаем сам файл по указанному пути
touch ~/.gitignore_global
nano ~/.gitignore_global (добавляем значения из файла, взятого по ссылке)

git remote add origin <URL> - связать локальный репозиторий с удаленным (remote - команда для управления удаленными репозиториями, origin - имя удаленного репозитория (по соглашению принято такое имя), <URL> берем с гитхаба)

git push -u origin master - отправить локальные коммиты на удаленный репозиторий (флаг -u связывает локальную ветку master с удаленной origin, после отправки этого пуша дальше можно просто писать git push)

git branch - проверить в какой ветке находимся

git clone <URL> - копировать гит-репозиторий к себе на комп (URL с гитхаба)
git clone --depth=5 <URL> - копировать последние 5 коммитов (указываем глубину репозитория, которую хотим иметь на данной локальной машине)
git clone <URL> <path> - копировать репозиторий в указанную папку (например, ~/git_repo)
git clone --bare <URL> - создать "голый" репозиторий (клонируем только данные гита без рабочей папки (индекса),  т.е. те файлы и папки, которые необходимы для управления версиями (самим репо)). Используется для создания центрального репозитория/или  бэкапов/или автоматизации процессов развертывания

git pull - подтянуть с гитхаба репозиторий на комп

ssh -T git@github.com - проверяет настроено ли SSH-подключение к github (-T - проверка подключения)
ssh-keygen -t ed25519 -C "email" - создаем пару криптографических ключей (публичный и приватный) с помощью алгоритма ed25519 и добавляем комментарий email к нему

- nano

Ctrl + O - сохранить в редакторе nano
Ctrl + X - выйти из редактора
Ctrl + K - вырезать строку
